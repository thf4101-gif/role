구글 스프레드시트 본부(Apps Script) 연동

구글 스프레드시트의 확장 프로그램 > Apps Script를 웹 앱(Web App)으로 배포한 URL을 입력하면 요원 명단 불러오기와 배정 결과 저장을 실시간으로 자동화할 수 있습니다.

※ 배포 설정: '액세스 권한: 모든 사용자(Anyone)'로 배포해야 웹 앱에서 즉시 접근 가능합니다.
현재 상태: 설정된 URL 없음 (브라우저 로컬 저장소 모드로 작동 중)
📋 Apps Script 예제 코드 보기 (스프레드시트에 복사해 넣기) 
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var students = [];
  var roles = [];
  for (var i = 1; i < data.length; i++) {
    if (data[i][0]) students.push(data[i][0]);
    if (data[i][1]) roles.push({
      title: data[i][1],
      icon: data[i][2] || '⭐',
      desc: data[i][3] || ''
    });
  }
  return ContentService.createTextOutput(JSON.stringify({
    success: true,
    students: students,
    roles: roles
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var params = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("히어로배정결과") || ss.insertSheet("히어로배정결과");
  sheet.clear();
  sheet.appendRow(["번호", "히어로 요원", "임무명", "임무 설명", "작전배정일시"]);
  var list = params.assignments || [];
  for (var i = 0; i < list.length; i++) {
    sheet.appendRow([i + 1, list[i].studentName, list[i].roleTitle, list[i].roleDesc, new Date()]);
  }
  return ContentService.createTextOutput(JSON.stringify({
    success: true,
    message: "성공적으로 저장되었습니다."
  })).setMimeType(ContentService.MimeType.JSON);
}
