Apps Script 연동 소스 코드 보기 
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var students = [];
  var roles = [];
  for (var i = 1; i < data.length; i++) {
    if (data[i][0]) students.push(data[i][0]);
    if (data[i][1]) roles.push({
      title: data[i][1],
      icon: data[i][2] || '✨',
      desc: data[i][3] || ''
    });
  }
  return ContentService.createTextOutput(JSON.stringify({
    success: true,
    students: students,
    roles: roles
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var params = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("역할배정결과") || ss.insertSheet("역할배정결과");
  sheet.clear();
  sheet.appendRow(["번호", "학생 이름", "역할 이름", "역할 설명", "배정일시"]);
  var list = params.assignments || [];
  for (var i = 0; i < list.length; i++) {
    sheet.appendRow([i + 1, list[i].studentName, list[i].roleTitle, list[i].roleDesc, new Date()]);
  }
  return ContentService.createTextOutput(JSON.stringify({
    success: true,
    message: "성공적으로 저장되었습니다."
  })).setMimeType(ContentService.MimeType.JSON);
}
