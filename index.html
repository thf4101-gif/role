구글 스프레드시트 (Apps Script) 연동 설정

구글 스프레드시트의 확장 프로그램 > Apps Script를 웹 앱(Web App)으로 배포한 URL을 등록하면 학생 명단 불러오기와 배정 결과 자동 저장이 원클릭으로 작동합니다.

배포 시 '액세스 권한: 모든 사용자(Anyone)'로 배포해야 웹에서 바로 통신할 수 있습니다.
현재 상태: 설정된 URL 없음 (브라우저 로컬 저장소 모드로 작동 중)
📋 Apps Script 예제 코드 보기 (스프레드시트에 복사해 넣기) 
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  // 첫 번째 행은 헤더(이름, 역할 등), 두 번째 행부터 데이터
  var students = [];
  var roles = [];
  for (var i = 1; i < data.length; i++) {
    if (data[i][0]) students.push(data[i][0]);
    if (data[i][1]) roles.push({
      title: data[i][1],
      icon: data[i][2] || '⭐',
      desc: data[i][3] || ''
    });
  }
  return ContentService.createTextOutput(JSON.stringify({
    success: true,
    students: students,
    roles: roles
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var params = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("배정결과") || ss.insertSheet("배정결과");
  sheet.clear();
  sheet.appendRow(["번호", "학생 이름", "역할 이름", "역할 설명", "배정일시"]);
  var list = params.assignments || [];
  for (var i = 0; i < list.length; i++) {
    sheet.appendRow([i + 1, list[i].studentName, list[i].roleTitle, list[i].roleDesc, new Date()]);
  }
  return ContentService.createTextOutput(JSON.stringify({
    success: true,
    message: "성공적으로 저장되었습니다."
  })).setMimeType(ContentService.MimeType.JSON);
}
